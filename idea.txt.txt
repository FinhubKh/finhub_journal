    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // --- DOM Elements ---
            const els = {
                capital: document.getElementById('capital'),
                targetReturn: document.getElementById('targetReturn'),
                returnLabel: document.getElementById('returnLabel'),
                frequency: document.getElementById('frequency'),
                growthType: document.getElementById('growthType'),
                timeHorizon: document.getElementById('timeHorizon'),
                timeLabel: document.getElementById('timeLabel'),
                timeUnit: document.getElementById('timeUnit'),
                btnSub10: document.getElementById('btnSub10'),
                btnSub1: document.getElementById('btnSub1'),
                btnAdd1: document.getElementById('btnAdd1'),
                btnAdd10: document.getElementById('btnAdd10'),
                globalTrades: document.getElementById('globalTrades'),
                globalWinRate: document.getElementById('globalWinRate'),
                globalRR: document.getElementById('globalRR'),
                hudFinal: document.getElementById('hudFinalBalance'),
                hudReturn: document.getElementById('hudTotalReturn'),
                scheduleBody: document.getElementById('scheduleBody'),
                ctx: document.getElementById('growthChart').getContext('2d')
            };

            let chartInstance = null;
            
            // Store custom risk overrides. Key = period number, Value = { riskAmt: number }
            let customRowData = {};

            // --- Currency Formatter ---
            const formatter = new Intl.NumberFormat('en-US', {
                style: 'currency',
                currency: 'USD',
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            });

            const pctFormatter = new Intl.NumberFormat('en-US', {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            });

            // --- STREAMING_CHUNK:Building the Time Horizon logic... ---
            // Unit conversion bounds
            const timeBounds = {
                weeks: { min: 1, max: 6240 }, // up to 120 years
                months: { min: 1, max: 120 }, // 10 years
                years: { min: 1, max: 30 }
            };

            function updateTimeSliderBounds() {
                const unit = els.timeUnit.value;
                const bounds = timeBounds[unit];
                els.timeHorizon.min = bounds.min;
                els.timeHorizon.max = bounds.max;
                
                // clamp current value to new bounds if necessary
                let currentVal = parseInt(els.timeHorizon.value);
                if (currentVal > bounds.max) els.timeHorizon.value = bounds.max;
                if (currentVal < bounds.min) els.timeHorizon.value = bounds.min;
            }

            function adjustTime(amount) {
                const current = parseInt(els.timeHorizon.value);
                const min = parseInt(els.timeHorizon.min);
                const max = parseInt(els.timeHorizon.max);
                let next = current + amount;
                
                if (next < min) next = min;
                if (next > max) next = max;
                
                els.timeHorizon.value = next;
                updateCalculator();
            }

            // --- STREAMING_CHUNK:Building the core Calculation Engine... ---
            function updateCalculator() {
                // Get inputs
                const principal = parseFloat(els.capital.value) || 0;
                const ratePct = parseFloat(els.targetReturn.value) || 0;
                const rateDecimal = ratePct / 100;
                const freq = els.frequency.value;
                const isCompound = els.growthType.value === 'compound';
                const timeValue = parseInt(els.timeHorizon.value) || 1;
                const unit = els.timeUnit.value;

                const globalTrades = parseInt(els.globalTrades.value) || 1;
                const globalWinRate = (parseFloat(els.globalWinRate.value) || 0) / 100;
                const globalRR = parseFloat(els.globalRR.value) || 1;

                // Update Labels
                els.returnLabel.textContent = `${ratePct}%`;
                els.timeLabel.textContent = `${timeValue} ${unit}`;

                // Calculate total periods (standardized to chart points)
                let totalPeriods = timeValue;
                if (unit === 'years') {
                    totalPeriods = freq === 'week' ? timeValue * 52 : timeValue * 12;
                } else if (unit === 'months' && freq === 'week') {
                    totalPeriods = timeValue * 4; // Approx 4 weeks per month
                } else if (unit === 'weeks' && freq === 'month') {
                    totalPeriods = Math.ceil(timeValue / 4);
                }

                // Protect against massive loops freezing browser
                if (totalPeriods > 6240) totalPeriods = 6240;

                const chartLabels = [];
                const chartData = [];
                let currentBalance = principal;
                
                let tableHTML = '';

                // Generate sequence
                for (let period = 1; period <= totalPeriods; period++) {
                    let startCap = currentBalance;
                    
                    // Determine risk directly from customRowData for this specific period
                    let reqRiskAmount;
                    if (customRowData[period]?.riskAmt !== undefined) {
                        reqRiskAmount = customRowData[period].riskAmt;
                    } else {
                        // Default logic if no override exists
                        let targetProfit = isCompound ? (startCap * rateDecimal) : (principal * rateDecimal);
                        let edge = (globalTrades * globalWinRate * globalRR) - (globalTrades * (1 - globalWinRate));
                        reqRiskAmount = edge > 0 ? (targetProfit / edge) : 10;
                    }
                    
                    let edgeMultiplier = (globalTrades * globalWinRate * globalRR) - (globalTrades * (1 - globalWinRate));
                    let targetProfit = edgeMultiplier > 0 ? (reqRiskAmount * edgeMultiplier) : 0;
                    let targetReturnPct = startCap > 0 ? (targetProfit / startCap) * 100 : 0;
                    
                    let riskStr = edgeMultiplier > 0 
                        ? `<div class="flex items-center justify-center gap-1">
                               <span class="text-slate-500 font-bold">$</span>
                               <input type="number" step="1" min="0" class="risk-input w-24 bg-slate-800 border ${customRowData[period]?.riskAmt !== undefined ? 'border-amber-500 text-amber-400' : 'border-slate-600 text-slate-200'} rounded px-2 py-1 text-sm font-mono focus:outline-none focus:border-blue-500 text-right transition-colors" data-period="${period}" value="${reqRiskAmount.toFixed(2)}">
                           </div>`
                        : `<span class="text-red-400 text-xs font-semibold">Invalid Edge</span>`;
                    
                    currentBalance += targetProfit;

                    chartLabels.push(`${freq === 'week' ? 'Wk' : 'Mo'} ${period}`);
                    chartData.push(currentBalance);

                    // Only render table rows for the first 500 periods to prevent DOM lag
                    if (period <= 500) {
                        tableHTML += `
                            <tr class="hover:bg-slate-800/50 transition-colors">
                                <td class="px-6 py-3 font-medium text-slate-300 border-r border-slate-800">${period}</td>
                                <td class="px-6 py-3 text-slate-300 border-r border-slate-800">${formatter.format(startCap)}</td>
                                <td class="px-6 py-3 border-r border-slate-800">${riskStr}</td>
                                <td class="px-6 py-3 text-right text-emerald-400 border-r border-slate-800">+${formatter.format(targetProfit)}</td>
                                <td class="px-6 py-3 text-right text-slate-400 border-r border-slate-800">${pctFormatter.format(targetReturnPct)}%</td>
                                <td class="px-6 py-3 text-right font-bold text-white">${formatter.format(currentBalance)}</td>
                            </tr>
                        `;
                    }
                }

                els.scheduleBody.innerHTML = tableHTML;
                attachTableListeners();

                // Update HUD
                els.hudFinal.textContent = formatter.format(currentBalance);
                const totalReturnPct = principal > 0 ? ((currentBalance - principal) / principal) * 100 : 0;
                els.hudReturn.textContent = `+${pctFormatter.format(totalReturnPct)}%`;

                renderChart(chartLabels, chartData);
            }

            // --- STREAMING_CHUNK:Setting up 10-week block cascade logic... ---
            function attachTableListeners() {
                // Listen for changes on Risk inputs (Dollar amounts)
                document.querySelectorAll('.risk-input').forEach(input => {
                    input.addEventListener('change', (e) => {
                        let period = parseInt(e.target.dataset.period);
                        let val = parseFloat(e.target.value);
                        
                        // Determine the end of the current 10-week block (10, 20, 30, etc.)
                        let blockEnd = Math.ceil(period / 10) * 10;
                        
                        if (isNaN(val) || e.target.value === '') {
                            // Clear overrides from this period to the end of the 10-week block
                            for(let i = period; i <= blockEnd; i++) {
                                if(customRowData[i]) delete customRowData[i].riskAmt;
                            }
                        } else {
                            // Cascade the exact dollar value down to the end of the current 10-week block
                            for(let i = period; i <= blockEnd; i++) {
                                if (!customRowData[i]) customRowData[i] = {};
                                customRowData[i].riskAmt = val;
                            }
                        }
                        updateCalculator();
                    });
                });
            }

            // --- STREAMING_CHUNK:Setting up Chart.js... ---
            function renderChart(labels, data) {
                if (chartInstance) {
                    chartInstance.destroy();
                }

                // Gradient Fill
                let gradient = els.ctx.createLinearGradient(0, 0, 0, 400);
                gradient.addColorStop(0, 'rgba(59, 130, 246, 0.4)'); // Blue
                gradient.addColorStop(1, 'rgba(59, 130, 246, 0.0)');

                chartInstance = new Chart(els.ctx, {
                    type: 'line',
                    data: {
                        labels: labels,
                        datasets: [{
                            label: 'Account Balance',
                            data: data,
                            borderColor: '#3b82f6', // Tailwind blue-500
                            backgroundColor: gradient,
                            borderWidth: 2,
                            pointRadius: 0,
                            pointHoverRadius: 6,
                            pointHoverBackgroundColor: '#10b981', // Tailwind emerald-500
                            fill: true,
                            tension: 0.4 // Smooth curves
                        }]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        interaction: {
                            intersect: false,
                            mode: 'index',
                        },
                        plugins: {
                            legend: { display: false },
                            tooltip: {
                                backgroundColor: 'rgba(15, 23, 42, 0.9)',
                                titleColor: '#94a3b8',
                                bodyColor: '#fff',
                                bodyFont: { family: 'monospace', size: 14 },
                                padding: 12,
                                borderColor: 'rgba(255,255,255,0.1)',
                                borderWidth: 1,
                                callbacks: {
                                    label: function(context) {
                                        return formatter.format(context.parsed.y);
                                    }
                                }
                            }
                        },
                        scales: {
                            x: {
                                grid: { color: 'rgba(255,255,255,0.05)' },
                                ticks: {
                                    color: '#64748b',
                                    maxTicksLimit: 10
                                }
                            },
                            y: {
                                grid: { color: 'rgba(255,255,255,0.05)' },
                                ticks: {
                                    color: '#64748b',
                                    callback: function(value) {
                                        if (value >= 1000000) return '$' + (value / 1000000).toFixed(1) + 'M';
                                        if (value >= 1000) return '$' + (value / 1000).toFixed(0) + 'k';
                                        return '$' + value;
                                    }
                                }
                            }
                        }
                    }
                });
            }

            // --- STREAMING_CHUNK:Attaching standard Event Listeners... ---
            els.capital.addEventListener('input', updateCalculator);
            els.targetReturn.addEventListener('input', updateCalculator);
            els.frequency.addEventListener('change', updateCalculator);
            els.growthType.addEventListener('change', updateCalculator);
            els.timeHorizon.addEventListener('input', updateCalculator);
            
            els.globalTrades.addEventListener('input', updateCalculator);
            els.globalWinRate.addEventListener('input', updateCalculator);
            els.globalRR.addEventListener('input', updateCalculator);

            els.timeUnit.addEventListener('change', () => {
                updateTimeSliderBounds();
                updateCalculator();
            });

            els.btnSub10.addEventListener('click', () => adjustTime(-10));
            els.btnSub1.addEventListener('click', () => adjustTime(-1));
            els.btnAdd1.addEventListener('click', () => adjustTime(1));
            els.btnAdd10.addEventListener('click', () => adjustTime(10));

            // Initialize
            updateTimeSliderBounds();
            updateCalculator();
        });
    </script>
